// Run as a statement block through Unity CLI eval in Edit mode.
// Original PNG pixels are never changed. Only the temporary collision mask is filled.
if (UnityEditor.EditorApplication.isPlaying) throw new System.Exception("Stop Play mode first.");
var outlineMethod = typeof(UnityEditor.Editor).Assembly.GetType("UnityEditor.Sprites.SpriteUtility")
    ?.GetMethod("GenerateOutline", System.Reflection.BindingFlags.Static | System.Reflection.BindingFlags.Public | System.Reflection.BindingFlags.NonPublic);
if (outlineMethod == null) throw new System.Exception("This Unity Editor has no outline generator.");
UnityEditor.AssetDatabase.Refresh();
System.IO.Directory.CreateDirectory("Assets/Game/Memes");
UnityEditor.AssetDatabase.Refresh();
var report = new System.Text.StringBuilder();
foreach (var guid in UnityEditor.AssetDatabase.FindAssets("t:Texture2D", new[] { "Assets/Images" }))
{
    var path = UnityEditor.AssetDatabase.GUIDToAssetPath(guid);
    if (!path.EndsWith(".png", System.StringComparison.OrdinalIgnoreCase)) continue;
    var importer = UnityEditor.AssetImporter.GetAtPath(path) as UnityEditor.TextureImporter;
    if (importer == null) throw new System.Exception("Missing texture importer: " + path);
    importer.textureType = UnityEditor.TextureImporterType.Sprite;
    importer.spriteImportMode = UnityEditor.SpriteImportMode.Single;
    importer.spritePixelsPerUnit = 100;
    importer.alphaIsTransparency = true;
    importer.mipmapEnabled = false;
    importer.maxTextureSize = 2048;
    importer.SaveAndReimport();
    var sprite = UnityEditor.AssetDatabase.LoadAssetAtPath<UnityEngine.Sprite>(path);
    if (sprite == null) throw new System.Exception("Sprite import failed: " + path);
    var texture = new UnityEngine.Texture2D(2, 2);
    UnityEngine.GameObject go = null;
    try
    {
        UnityEngine.ImageConversion.LoadImage(texture, System.IO.File.ReadAllBytes(path));
        int w = texture.width, h = texture.height;
        var pixels = texture.GetPixels32();
        var mask = new bool[w * h];
        int minX = w, minY = h, maxX = 0, maxY = 0;
        for (int y = 0; y < h; y++) for (int x = 0; x < w; x++)
        {
            if (pixels[y * w + x].a < 32) continue;
            mask[y * w + x] = true;
            minX = System.Math.Min(minX, x); maxX = System.Math.Max(maxX, x);
            minY = System.Math.Min(minY, y); maxY = System.Math.Max(maxY, y);
        }
        if (minX > maxX) throw new System.Exception("Empty image: " + path);
        // These cropped/open sketches need explicit closing edges. Coordinates refer to
        // the original PNG (top-left origin); only the collision mask receives the edges.
        System.Action<int, int, int, int> close = (x0, y0, x1, y1) =>
        {
            int steps = System.Math.Max(System.Math.Abs(x1 - x0), System.Math.Abs(y1 - y0));
            for (int i = 0; i <= steps; i++)
            {
                int x = x0 + (x1 - x0) * i / System.Math.Max(1, steps);
                int y = h - 1 - (y0 + (y1 - y0) * i / System.Math.Max(1, steps));
                if (x >= 0 && x < w && y >= 0 && y < h) mask[y * w + x] = true;
            }
        };
        if (path.EndsWith("1_상동호수공원.png"))
        {
            close(543, 31, 810, 34); close(391, 347, 413, 343);
            close(94, 459, 271, 661); close(271, 661, 300, 844);
            close(300, 844, 831, 844); close(1363, 781, 1075, 838);
        }
        if (path.EndsWith("1_인직이.png")) close(530, 820, 1065, 815);
        if (path.EndsWith("2_셀카.png")) { close(731, 900, 1031, 887); close(1145, 443, 1145, 457); }
        // Close tiny breaks in hand-drawn outlines using a 9-pixel square kernel.
        // Separable prefix sums keep this linear in the number of pixels.
        for (int pass = 0; pass < 4; pass++)
        {
            bool horizontal = pass % 2 == 0, dilate = pass < 2;
            int length = horizontal ? w : h, lines = horizontal ? h : w;
            var next = new bool[mask.Length];
            var prefix = new int[length + 1];
            for (int line = 0; line < lines; line++)
            {
                prefix[0] = 0;
                for (int i = 0; i < length; i++) prefix[i + 1] = prefix[i] + (mask[horizontal ? line * w + i : i * w + line] ? 1 : 0);
                for (int i = 0; i < length; i++)
                {
                    int lo = System.Math.Max(0, i - 4), hi = System.Math.Min(length, i + 5);
                    int count = prefix[hi] - prefix[lo];
                    next[horizontal ? line * w + i : i * w + line] = dilate ? count > 0 : count == hi - lo;
                }
            }
            mask = next;
        }
        // Flood only exterior transparency. Enclosed eyes, torso and wheels become solid.
        var outside = new bool[mask.Length];
        var queue = new System.Collections.Generic.Queue<int>();
        System.Action<int> visit = i => { if (!mask[i] && !outside[i]) { outside[i] = true; queue.Enqueue(i); } };
        for (int x = 0; x < w; x++) { visit(x); visit((h - 1) * w + x); }
        for (int y = 0; y < h; y++) { visit(y * w); visit(y * w + w - 1); }
        while (queue.Count > 0)
        {
            int i = queue.Dequeue(), x = i % w, y = i / w;
            if (x > 0) visit(i - 1); if (x < w - 1) visit(i + 1);
            if (y > 0) visit(i - w); if (y < h - 1) visit(i + w);
        }
        for (int i = 0; i < pixels.Length; i++) pixels[i] = new UnityEngine.Color32(255, 255, 255, outside[i] ? (byte)0 : (byte)255);
        texture.SetPixels32(pixels); texture.Apply();
        var args = new object[] { texture, new UnityEngine.Rect(0, 0, w, h), .08f, (byte)32, false, null };
        outlineMethod.Invoke(null, args);
        var generated = (UnityEngine.Vector2[][])args[5];
        var paths = new System.Collections.Generic.List<UnityEngine.Vector2[]>();
        foreach (var points in generated)
        {
            float area = 0;
            for (int i = 0; i < points.Length; i++) { var a = points[i]; var b = points[(i + 1) % points.Length]; area += a.x * b.y - b.x * a.y; }
            if (points.Length >= 3 && System.Math.Abs(area) >= 48) paths.Add(points);
        }
        if (paths.Count == 0) throw new System.Exception("No valid outline: " + path);
        var center = new UnityEngine.Vector2((minX + maxX + 1) * .5f - w * .5f, (minY + maxY + 1) * .5f - h * .5f);
        float scale = 1.4f / (maxY - minY + 1);
        go = new UnityEngine.GameObject(System.IO.Path.GetFileNameWithoutExtension(path));
        var art = new UnityEngine.GameObject("Art"); art.transform.SetParent(go.transform, false);
        art.transform.localPosition = -center * scale;
        art.transform.localScale = UnityEngine.Vector3.one * (100 * scale);
        var render = art.AddComponent<UnityEngine.SpriteRenderer>(); render.sprite = sprite; render.sortingOrder = 5;
        var collider = go.AddComponent<UnityEngine.PolygonCollider2D>();
        collider.pathCount = paths.Count; collider.isTrigger = true;
        int vertices = 0;
        for (int p = 0; p < paths.Count; p++)
        {
            var points = paths[p]; vertices += points.Length;
            for (int i = 0; i < points.Length; i++) points[i] = (points[i] - center) * scale;
            collider.SetPath(p, points);
        }
        go.AddComponent<MemeDodge.Danger>();
        UnityEngine.Physics2D.SyncTransforms();
        if (collider.bounds.size.y > 1.6f || collider.bounds.size.y < 1f) throw new System.Exception("Unexpected collider bounds: " + path);
        var prefabPath = "Assets/Game/Memes/" + go.name + ".prefab";
        UnityEditor.PrefabUtility.SaveAsPrefabAsset(go, prefabPath);
        report.AppendLine(go.name + ": " + paths.Count + " paths, " + vertices + " vertices, bounds " + collider.bounds.size);
    }
    finally { if (go != null) UnityEngine.Object.DestroyImmediate(go); UnityEngine.Object.DestroyImmediate(texture); }
}
UnityEditor.AssetDatabase.SaveAssets();
System.IO.Directory.CreateDirectory("../tmp/verification");
System.IO.File.WriteAllText("../tmp/verification/meme-colliders.txt", report.ToString());
return report.ToString();
